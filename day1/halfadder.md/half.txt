// Code your design here
module Half_Adder(
    input A,
    input B,
    output Sum,
    output Carry
);

assign Sum   = A ^ B;   // XOR for Sum
assign Carry = A & B;   // AND for Carry

endmodule


// Code your testbench here
module Half_Adder_TB;

reg A;
reg B;
wire Sum;
wire Carry;

// Instantiate the Half Adder
  Half_Adder DUT(A,B,Sum,Carry
);

initial
begin
    // Apply test inputs
    A=0; B=0; #10;
    A=0; B=1; #10;
    A=1; B=0; #10;
    A=1; B=1; #10;
end
initial
begin
    $dumpfile("dump.vcd");        // ⭐ create waveform file
    $dumpvars(0, Half_Adder_TB);  // ⭐ record signals
end
endmodule
